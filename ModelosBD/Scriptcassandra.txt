-- Script CQL ejecutable: reproduce keyspace y tablas que crean
-- CassandraSingleton, CrearTablasMonitoreo y CrearTablaAlertas.
-- Coincide con los tipos, nombres y claves de DatosDePruebaCassandra e
-- InsertarCassandraMonitoreo. IF NOT EXISTS no migra tablas ya existentes.
-- La aplicación genera 60 lecturas aleatorias entre 2026-09-01 y 2026-10-06,
-- con IDs reales de sensores/contenedores MongoDB. Por eso esta fila fija es
-- una muestra ejecutable del mismo formato y no reemplaza aquella carga.
CREATE KEYSPACE IF NOT EXISTS logistica
WITH replication = {'class': 'SimpleStrategy', 'replication_factor': 1};

CREATE TABLE IF NOT EXISTS logistica.lecturas_sensor (
sensor_id text,
fecha_dia date,
fecha_hora timestamp,
contenedor_id text,

temperatura decimal,
humedad decimal,
vibracion decimal,

latitud decimal,
longitud decimal,

bateria decimal,

pais text,
region text,

PRIMARY KEY ((sensor_id, fecha_dia), fecha_hora)
)
WITH CLUSTERING ORDER BY (fecha_hora DESC);

CREATE TABLE IF NOT EXISTS logistica.metricas_iot_region_dia (
region text,
fecha_dia date,

temperatura_min decimal,
temperatura_max decimal,

humedad_promedio decimal,

cantidad_lecturas bigint,

PRIMARY KEY ((region), fecha_dia)
)
WITH CLUSTERING ORDER BY (fecha_dia DESC);

CREATE TABLE IF NOT EXISTS logistica.metricas_iot_pais_dia (
pais text,
fecha_dia date,

humedad_promedio decimal,
temperatura_promedio decimal,

cantidad_lecturas bigint,

PRIMARY KEY ((pais), fecha_dia)
)
WITH CLUSTERING ORDER BY (fecha_dia DESC);

CREATE TABLE IF NOT EXISTS logistica.posiciones_sensor (
sensor_id text,
fecha_hora timestamp,
contenedor_id text,
latitud decimal,
longitud decimal,
pais text,
region text,
PRIMARY KEY ((sensor_id), fecha_hora)
)
WITH CLUSTERING ORDER BY (fecha_hora DESC);

CREATE TABLE IF NOT EXISTS logistica.alertas (
id text PRIMARY KEY,
nombre text,
descripcion text
);
