 * Los envíos se almacenan en Neo4j.
 */

const base = db.getSiblingDB("datos2");

// ========================================
// COLECCIONES
// ========================================

const colecciones = [
    "clientes",
    "usuarios",
    "roles",
    "contenedores",
    "vehiculos",
    "sensores",
    "eventos_logisticos",
    "incidentes",
    "alertas",
    "reportes",
    "facturas",
    "pagos",
    "cuentas_corrientes",
    "movimientos_cuenta_corriente",
    "aperturas_cuenta_corriente",
    "cierres_mensuales_cuenta_corriente",
    "contadores",
    "conversaciones",
    "mensajes"
];

const existentes = new Set(base.getCollectionNames());

for (const nombre of colecciones) {
    if (!existentes.has(nombre)) {
        base.createCollection(nombre);
        print("Creada: " + nombre);
    } else {
        print("Ya existe: " + nombre);
    }
}

// ========================================
// ÍNDICE ÚNICO - APERTURAS
// ========================================

base.aperturas_cuenta_corriente.createIndex(
    {
        cuenta_corriente_id: 1,
        anio: 1,
        mes: 1
    },
    {
        unique: true
    }
);

// ========================================
// ÍNDICE ÚNICO - CIERRES
// ========================================

base.cierres_mensuales_cuenta_corriente.createIndex(
    {
        cuenta_corriente_id: 1,
        anio: 1,
        mes: 1
    },
    {
        unique: true
    }
);

// ========================================
// ÍNDICE ÚNICO PARCIAL - MOVIMIENTOS
// ========================================

base.movimientos_cuenta_corriente.createIndex(
    {
        cuenta_corriente_id: 1,
        tipo: 1,
        referencia_id: 1
    },
    {
        unique: true,
        partialFilterExpression: {
            referencia_id: {
                $type: "string"
            }
        }
    }
);

// ========================================
// VERIFICACIÓN
// ========================================

print("\nCOLECCIONES E ÍNDICES:");

for (const nombre of colecciones) {
    print("\n" + nombre);
    printjson(base.getCollection(nombre).getIndexes());
}

print("\nScript de MongoDB finalizado.");
