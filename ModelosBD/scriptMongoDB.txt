// Creación de las colecciones que escriben los DAO actuales.
// Base documental: datos2. Ejecutar con mongosh --file ModelosBD/scriptMongoDB.txt.
// No contiene datos de ejemplo: la carga coherente entre motores está en
// DatosPorDefecto/GeneradorDatosPruebaMongo.java y DatosPruebaTotal.java.
// No ejecutar junto con Main esperando conservarlos: Main borra las bases al iniciar.
// Los DAO no crean índices secundarios ni validadores; este script tampoco.
// El índice estándar de _id queda a cargo de MongoDB.
const base = db.getSiblingDB("datos2");
const colecciones = [
  "clientes",
  "usuarios",
  "roles",
  "contenedores",
  "vehiculos",
  "sensores",
  "eventos_logisticos",
  "incidentes",
  "alertas",
  "reportes",
  "facturas",
  "pagos",
  "cuentas_corrientes",
  "conversaciones",
  "mensajes"
];
const existentes = new Set(base.getCollectionNames());
for (const nombre of colecciones) {
    if (!existentes.has(nombre)) base.createCollection(nombre);
}
// No crear envios: se persisten como nodos y relaciones en Neo4j.
// Consultas de inspección, sin agregar restricciones:
for (const nombre of colecciones) print(nombre + ": " + JSON.stringify(base.getCollection(nombre).getIndexes()));
